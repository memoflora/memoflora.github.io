{{- $id := .Get 0 -}} {{- $url := printf "https://open.spotify.com/album/%s" $id -}} {{- $api := printf "https://open.spotify.com/oembed?url=%s" ($url | urlquery) -}} {{- with try (resources.GetRemote $api) -}} {{- with .Err -}} {{- errorf "spotify-cover %s: %s" $id . -}} {{- else with .Value -}} {{- $meta := .Content | transform.Unmarshal -}} {{- with try (resources.GetRemote $meta.thumbnail_url) -}} {{- with .Err -}} {{- errorf "spotify-cover %s: %s" $id . -}} {{- else with .Value -}} {{- else -}} {{- errorf "spotify-cover %s: cover image request returned nothing (%s)" $id $meta.thumbnail_url -}} {{- end -}} {{- end -}} {{- else -}} {{- errorf "spotify-cover %s: oEmbed request returned nothing (%s)" $id $api -}} {{- end -}} {{- end -}}
